cmake_minimum_required(VERSION 3.21)
project(LimbiQt VERSION 0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(LimbiQt WIN32 main.cpp)
qt_add_resources(LimbiQt content PREFIX "/" FILES content.json)
target_link_libraries(LimbiQt PRIVATE Qt6::Widgets Qt6::Network)
add_executable(ContentTests content-tests.cpp)
target_link_libraries(ContentTests PRIVATE Qt6::Core Qt6::Network)
enable_testing()
add_test(NAME content-validation COMMAND ContentTests)
qt_add_executable(LimbiPublisher WIN32 publisher.cpp)
target_link_libraries(LimbiPublisher PRIVATE Qt6::Widgets Qt6::Network)
